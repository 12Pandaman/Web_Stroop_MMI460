การเชื่อมต่อ Google Sheets แบบอัตโนมัติ (Google Apps Script Web App)

วิธีที่ง่ายที่สุด สะดวกที่สุด และไม่ต้องติดตั้ง Database หรือ Server เสียค่าใช้จ่ายใดๆ ทั้งสิ้น

ยังไม่ได้เชื่อมต่อ URL *บันทึกไว้ในเบราว์เซอร์อัตโนมัติ ทุกครั้งที่มีการทดสอบจบ ข้อมูลจะถูกยิงเข้า Google Sheet ทันที

วิธีเปิดใช้งาน Google Apps Script (ทำเพียงครั้งเดียว ภายใน 2 นาที):

  1. สร้าง Google Sheet ใหม่ ➔ ตั้งชื่อว่า Stroop_Research_Data
  2. ไปที่เมนู ส่วนขยาย (Extensions) ➔ เลือก Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดด้านล่างนี้ลงไป: 
    function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Stroop_Data') || ss.getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'วันที่/เวลา', 'ผู้เข้าร่วม', 
      'Pre Congruent (ms)', 'Pre Incongruent (ms)', 'Pre ผลต่าง D (ms)',
      'Post Congruent (ms)', 'Post Incongruent (ms)', 'Post ผลต่าง D (ms)',
      'ผลต่าง Delta D (Post - Pre)', 'Pre ความถูกต้อง', 'Post ความถูกต้อง'
    ]);
  }
  
  if (payload.action === 'add_record' && payload.record) {
    var r = payload.record;
    sheet.appendRow([
      r.dateTimeStr || new Date().toLocaleString('th-TH'),
      r.participantName,
      r.preCongMs, r.preIncongMs, r.preDiffD,
      r.postCongMs, r.postIncongMs, r.postDiffD,
      r.deltaD,
      r.preAccuracy + '%', r.postAccuracy + '%'
    ]);
  } else if (payload.action === 'sync_all' && payload.records) {
    payload.records.forEach(function(r) {
      sheet.appendRow([
        r.dateTimeStr || new Date().toLocaleString('th-TH'),
        r.participantName,
        r.preCongMs, r.preIncongMs, r.preDiffD,
        r.postCongMs, r.postIncongMs, r.postDiffD,
        r.deltaD,
        r.preAccuracy + '%', r.postAccuracy + '%'
      ]);
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
  4. กดปุ่ม ทำให้ใช้งานได้ (Deploy) มุมบนขวา ➔ เลือก การทำให้ใช้งานได้ใหม่ (New deployment) ➔ เลือกประเภทเป็น เว็บแอป (Web app) ➔ ในช่อง "ผู้มีสิทธิ์เข้าถึง (Who has access)" ให้เลือกเป็น "ทุกคน (Anyone)" ➔ กด Deploy แล้วนำ URL ที่ได้มาวางในช่องด้านบน!
